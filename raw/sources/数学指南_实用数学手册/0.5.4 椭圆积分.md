$$
F (k, \varphi) = \int_ {0} ^ {\varphi} \frac {\mathrm{d} \psi}{\sqrt {1 - k ^ {2} \sin^ {2} \psi}} = \int_ {0} ^ {\sin \varphi} \frac {\mathrm{d} x}{\sqrt {1 - x ^ {2}} \sqrt {1 - k ^ {2} x ^ {2}}},
$$

$$
E (k, \varphi) = \int_ {0} ^ {\varphi} \sqrt {1 - k ^ {2} \sin^ {2} \psi} \mathrm{d} \psi = \int_ {0} ^ {\sin \varphi} \sqrt {\frac {1 - k ^ {2} x ^ {2}}{1 - x ^ {2}}} \mathrm{d} x,
$$

$$
K = F \bigg (k, \frac {\pi}{2} \bigg) = \int_ {0} ^ {\pi / 2} \frac {\mathrm{d} \psi}{\sqrt {1 - k ^ {2} \sin^ {2} \psi}} = \int_ {0} ^ {1} \frac {\mathrm{d} x}{\sqrt {1 - x ^ {2}} \sqrt {1 - k ^ {2} x ^ {2}}},
$$

$$
\mathrm{E} = E \left(k, \frac {\pi}{2}\right) = \int_ {0} ^ {\pi / 2} \sqrt {1 - k ^ {2} \sin^ {2} \psi} \mathrm{d} \psi = \int_ {0} ^ {1} \sqrt {\frac {1 - k ^ {2} x ^ {2}}{1 - x ^ {2}}} \mathrm{d} x,
$$

$$
4 (n + 1) ^ {2} \int \mathrm{E} x ^ {n} \mathrm{d} x - (2 n + 3) (2 n + 5) \int \mathrm{E} x ^ {n + 1} \mathrm{d} x
$$

$$
= (2 n + 3) ^ {2} \int \mathrm{K} x ^ {n + 1} \mathrm{d} x - 4 (n + 1) ^ {2} \int \mathrm{K} x ^ {n} \mathrm{d} x = 2 x ^ {n + 1} (\mathrm{E} - (2 n + 3) (1 - x) \mathrm{K}).
$$
